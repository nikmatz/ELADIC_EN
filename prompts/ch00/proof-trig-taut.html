1. What is proved: the statement in the book and in Lean

  • $\sin^2\theta+\cos^2\theta=1$
    theorem pythagorean (θ : ℝ) : sin θ ^ 2 + cos θ ^ 2 = 1
  • $1+\tan^2\theta$${}=\dfrac{1}{\cos^2\theta}$
    theorem one_add_tan_sq (θ : ℝ) (h : cos θ ≠ 0) : 1 + tan θ ^ 2 = 1 / cos θ ^ 2
  • $\sin(\alpha+\beta)$${}=\sin\alpha\cos\beta+\cos\alpha\sin\beta$
    theorem sin_of_sum (α β : ℝ) : sin (α + β) = sin α * cos β + cos α * sin β
  • $\sin(\alpha-\beta)$${}=\sin\alpha\cos\beta-\cos\alpha\sin\beta$
    theorem sin_of_diff (α β : ℝ) : sin (α - β) = sin α * cos β - cos α * sin β
  • $\cos(\alpha+\beta)$${}=\cos\alpha\cos\beta-\sin\alpha\sin\beta$
    theorem cos_of_sum (α β : ℝ) : cos (α + β) = cos α * cos β - sin α * sin β
  • $\cos(\alpha-\beta)$${}=\cos\alpha\cos\beta+\sin\alpha\sin\beta$
    theorem cos_of_diff (α β : ℝ) : cos (α - β) = cos α * cos β + sin α * sin β
  • $\sin 2\alpha=2\sin\alpha\cos\alpha$
    theorem sin_double (α : ℝ) : sin (2 * α) = 2 * sin α * cos α
  • $\cos 2\alpha$${}=\cos^2\alpha-\sin^2\alpha$
    theorem cos_double (α : ℝ) : cos (2 * α) = cos α ^ 2 - sin α ^ 2

In Mathlib, sine and cosine are defined by power series, not from the unit circle, and these identities are proved from the definition. For $1+\tan^2\theta$ we assume $\cos\theta\neq 0$, as in the book.

2. Where it is in Mathlib

The basic identities are available ready-made:

  • Real.sin_sq_add_cos_sq: $\sin^2x+\cos^2x=1$.
  • Real.sin_add: the sine of a sum.
  • Real.cos_add: the cosine of a sum.
  • Real.sin_two_mul: $\sin 2x=2\sin x\cos x$.
  • Real.cos_two_mul: $\cos 2x=2\cos^2x-1$.

import Mathlib

namespace Ch0.TrigTautotites
open Real

theorem pythagorean (θ : ℝ) : sin θ ^ 2 + cos θ ^ 2 = 1 := sin_sq_add_cos_sq θ

theorem one_add_tan_sq (θ : ℝ) (h : cos θ ≠ 0) : 1 + tan θ ^ 2 = 1 / cos θ ^ 2 := by
  have h2 : cos θ ^ 2 ≠ 0 := pow_ne_zero 2 h
  rw [tan_eq_sin_div_cos, div_pow, eq_div_iff h2, add_mul, one_mul, div_mul_cancel₀ _ h2]
  linarith [sin_sq_add_cos_sq θ]

theorem sin_of_sum (α β : ℝ) : sin (α + β) = sin α * cos β + cos α * sin β := sin_add α β

theorem sin_of_diff (α β : ℝ) : sin (α - β) = sin α * cos β - cos α * sin β := sin_sub α β

theorem cos_of_sum (α β : ℝ) : cos (α + β) = cos α * cos β - sin α * sin β := cos_add α β

theorem cos_of_diff (α β : ℝ) : cos (α - β) = cos α * cos β + sin α * sin β := cos_sub α β

theorem sin_double (α : ℝ) : sin (2 * α) = 2 * sin α * cos α := sin_two_mul α

theorem cos_double (α : ℝ) : cos (2 * α) = cos α ^ 2 - sin α ^ 2 := by
  rw [cos_two_mul]
  linear_combination sin_sq_add_cos_sq α

end Ch0.TrigTautotites
