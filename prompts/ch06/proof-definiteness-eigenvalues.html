1. What is proved: the statement in the book and in Lean

  • Positive definite means: $A=A^{T}$ and $\mathbf{x}^{T}A\mathbf{x}\gt 0$ for every $\mathbf{x}\ne\mathbf{0}$
    theorem pos_def_quadratic (A : Matrix (Fin n) (Fin n) ℝ) :
    A.PosDef ↔ A.IsHermitian ∧ ∀ ⦃x : Fin n → ℝ⦄, x ≠ 0 → 0 < x ⬝ᵥ (A *ᵥ x)
  • Positive definite $\Leftrightarrow\lambda_i\gt 0$ for every $i$
    theorem pos_def_iff (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) :
    A.PosDef ↔ ∀ i, 0 < hA.eigenvalues i
  • Positive semidefinite $\Leftrightarrow\lambda_i\geq 0$ for every $i$
    theorem pos_semidef_iff (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) :
    A.PosSemidef ↔ ∀ i, 0 ≤ hA.eigenvalues i
  • Negative definite $\Leftrightarrow\lambda_i\lt 0$ for every $i$
    theorem neg_def_iff (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) :
    (-A).PosDef ↔ ∀ i, hA.eigenvalues i < 0
  • Negative semidefinite $\Leftrightarrow\lambda_i\leq 0$ for every $i$
    theorem neg_semidef_iff (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) :
    (-A).PosSemidef ↔ ∀ i, hA.eigenvalues i ≤ 0
  • Indefinite $\Leftrightarrow$ mixed signs: there is a $\lambda_i\lt 0$ and there is a $\lambda_j\gt 0$
    theorem indefinite_iff (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) :
    (¬ A.PosSemidef ∧ ¬ (-A).PosSemidef) ↔
      (∃ i, hA.eigenvalues i < 0) ∧ (∃ i, 0 < hA.eigenvalues i)
  • Auxiliary: the eigenvalues of $-A$ are the negatives of the eigenvalues of $A$
    theorem neg_eig_exists (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) (i : Fin n) :
    ∃ j, hA.neg.eigenvalues j = -hA.eigenvalues i

The symmetric matrix is given by the hypothesis hA : A.IsHermitian: for real matrices, Hermitian means exactly $A^{T}=A$. Its $n$ real eigenvalues are hA.eigenvalues i. “Positive definite” is A.PosDef and “positive semidefinite” is A.PosSemidef. $A$ is negative definite or semidefinite exactly when $-A$ is positive definite or semidefinite, because $\mathbf{x}^{T}(-A)\mathbf{x}$${}=-\mathbf{x}^{T}A\mathbf{x}$; this is why the negative cases are written (-A).PosDef and (-A).PosSemidef. A matrix is indefinite when it is neither positive nor negative semidefinite, that is, when the form $\mathbf{x}^{T}A\mathbf{x}$ takes both positive and negative values.

2. Where it is in Mathlib

We used the following theorems:

  • Matrix.IsHermitian.posDef_iff_eigenvalues_pos: positive definite exactly when all eigenvalues are positive.
  • Matrix.IsHermitian.posSemidef_iff_eigenvalues_nonneg: positive semidefinite exactly when all eigenvalues are nonnegative.
  • Matrix.posDef_iff_dotProduct_mulVec: the definition via the quadratic form $\mathbf{x}^{T}A\mathbf{x}$.
  • Matrix.IsHermitian.spectrum_real_eq_range_eigenvalues: the spectrum consists exactly of the eigenvalues.
  • spectrum.neg_eq: the spectrum of $-A$ consists of the negatives of the elements of the spectrum of $A$.

import Mathlib

namespace Ch6.Oristikotita
open Matrix

variable {n : ℕ}

theorem pos_def_quadratic (A : Matrix (Fin n) (Fin n) ℝ) :
    A.PosDef ↔ A.IsHermitian ∧ ∀ ⦃x : Fin n → ℝ⦄, x ≠ 0 → 0 < x ⬝ᵥ (A *ᵥ x) := by
  rw [posDef_iff_dotProduct_mulVec]
  simp only [star_trivial]

theorem pos_def_iff (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) :
    A.PosDef ↔ ∀ i, 0 < hA.eigenvalues i :=
  hA.posDef_iff_eigenvalues_pos

theorem pos_semidef_iff (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) :
    A.PosSemidef ↔ ∀ i, 0 ≤ hA.eigenvalues i := by
  rw [hA.posSemidef_iff_eigenvalues_nonneg, Pi.le_def]
  simp only [Pi.zero_apply]

-- the eigenvalues of -A are the negatives of the eigenvalues of A
theorem neg_eig_exists (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) (i : Fin n) :
    ∃ j, hA.neg.eigenvalues j = -hA.eigenvalues i := by
  have h : -hA.eigenvalues i ∈ spectrum ℝ (-A) := by
    rw [← spectrum.neg_eq, Set.neg_mem_neg]
    exact hA.eigenvalues_mem_spectrum_real i
  rwa [hA.neg.spectrum_real_eq_range_eigenvalues, Set.mem_range] at h

theorem eig_exists_neg (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) (i : Fin n) :
    ∃ j, hA.eigenvalues j = -hA.neg.eigenvalues i := by
  have h : hA.neg.eigenvalues i ∈ spectrum ℝ (-A) := hA.neg.eigenvalues_mem_spectrum_real i
  rwa [← spectrum.neg_eq, Set.mem_neg, hA.spectrum_real_eq_range_eigenvalues, Set.mem_range] at h

theorem neg_def_iff (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) :
    (-A).PosDef ↔ ∀ i, hA.eigenvalues i < 0 := by
  rw [pos_def_iff (-A) hA.neg]
  constructor
  · intro h i
    obtain ⟨j, hj⟩ := neg_eig_exists A hA i
    have := h j
    linarith
  · intro h i
    obtain ⟨j, hj⟩ := eig_exists_neg A hA i
    have := h j
    linarith

theorem neg_semidef_iff (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) :
    (-A).PosSemidef ↔ ∀ i, hA.eigenvalues i ≤ 0 := by
  rw [pos_semidef_iff (-A) hA.neg]
  constructor
  · intro h i
    obtain ⟨j, hj⟩ := neg_eig_exists A hA i
    have := h j
    linarith
  · intro h i
    obtain ⟨j, hj⟩ := eig_exists_neg A hA i
    have := h j
    linarith

theorem indefinite_iff (A : Matrix (Fin n) (Fin n) ℝ) (hA : A.IsHermitian) :
    (¬ A.PosSemidef ∧ ¬ (-A).PosSemidef) ↔
      (∃ i, hA.eigenvalues i < 0) ∧ (∃ i, 0 < hA.eigenvalues i) := by
  rw [pos_semidef_iff A hA, neg_semidef_iff A hA]
  simp only [not_forall, not_le]

end Ch6.Oristikotita
