/* ============================================================
   maxima_ttl.mac
   Chapter 15 — Fundamental Theorem of Calculus (FTC)
                 The area function and the Leibniz rule
   Matzakos, N. (2026). Elements of Linear Algebra, Differential & Integral Calculus. NewTech Publications.
   ============================================================

   COMMANDS USED:
     assume(x > 0)             → sign assumption (FIRST OF ALL!)
     integrate(f, t, 0, x)     → area function F(x) = ∫₀ˣ f(t)dt
     diff(F, x)                → F'(x) — Part I of the FTC
     subst(b, x, F)            → substitution x := b (Newton–Leibniz formula)
     'integrate(f, t, a, b)    → UNEVALUATED integral; diff applies
                                 the Leibniz rule to it directly
     solve(eq, c)              → solving for the c of the MVT
     ratsimp, trigsimp, float  → simplification and decimal values
     plot2d([...], [x, a, b])  → plot

   CAUTION: assume(x > 0) must come BEFORE every integrate whose upper
   limit is x. Otherwise Maxima asks "Is x positive, negative or zero?"
   and in batch mode it blocks forever.

   Book activity:
     (a) F(x)=∫₀ˣf(t)dt for f=t², sin t, 1/(1+t²); check F'(x)=f(x)
     (b) Newton–Leibniz: ∫₁³(2x+1), ∫₀^(pi/2)cos x, ∫₁^e (ln x)/x
         first as F(b)-F(a) with subst, then directly
     (c) the Leibniz rule with variable limits
     (d) MVT for Integrals for f(x)=x² on [0,3]

   Run with: wxMaxima (File → Batch File) or
                 maxima -b maxima_ttl.mac
   ============================================================ */

/* FIRST OF ALL: the sign assumptions — no questions afterwards */
assume(x > 0)$
print("Assumptions in force:", facts())$

/* ── A. (a) Part I of the FTC: the area function ──────── */
print("")$
print("════ A. (a) Area function F(x) = ∫₀ˣ f(t)dt ════")$

cases_a : [t^2, sin(t), 1/(1+t^2)]$

for ft in cases_a do
  block([F, dF, f_at_x],
    F      : integrate(ft, t, 0, x),
    dF     : trigsimp(ratsimp(diff(F, x))),
    f_at_x : subst(x, t, ft),
    print("  f(t) =", ft),
    print("     F(x) = ∫₀ˣ f(t)dt =", F),
    print("     F'(x) =", dF, "     F'(x) - f(x) =",
          ratsimp(trigsimp(dF - f_at_x)), "  (must be 0)"))$

print("")$
print("  [Expected: f=t² → F=x³/3 ; f=sin t → F=1-cos x ;")$
print("               f=1/(1+t²) → F=atan(x)]")$

/* Values of the area function (they show the "accumulation") */
print("")$
F_sin : integrate(sin(t), t, 0, x)$
print("Accumulation for f(t)=sin t:")$
for a in [%pi/4, %pi/2, %pi, 3*%pi/2, 2*%pi] do
  print("   F(", a, ") =", ratsimp(subst(a, x, F_sin)),
        "=", float(subst(a, x, F_sin)))$
print("  [Maximum at x=pi (there f=sin x changes sign): F(pi)=2]")$

/* ── B. (b) Newton–Leibniz formula: ∫_a^b f = F(b) - F(a) ── */
print("")$
print("════ B. (b) Newton–Leibniz — two passes ════")$
print("First as F(b)-F(a) with subst, then directly with integrate(f,x,a,b).")$

/* Data from the book: [f(x), a, b, expected value] */
nl_cases : [[2*x + 1,        1,     3,      10],
            [cos(x),         0,     %pi/2,  1],
            [log(x)/x,       1,     %e,     1/2]]$

for cs in nl_cases do
  block([fx, aa, bb, expect, G, val1, val2],
    fx     : cs[1],
    aa     : cs[2],
    bb     : cs[3],
    expect : cs[4],
    G      : integrate(fx, x),                       /* antiderivative */
    val1   : ratsimp(subst(bb, x, G) - subst(aa, x, G)),   /* F(b)-F(a) */
    val2   : integrate(fx, x, aa, bb),                     /* directly  */
    print("  f(x) =", fx, "   on [", aa, ",", bb, "]"),
    print("     Antiderivative F(x) =", G),
    print("     (1) F(b) - F(a) =", val1, "=", float(val1)),
    print("     (2) integrate(f,x,a,b) =", val2, "=", float(val2)),
    print("     Do they agree? difference =", ratsimp(val1 - val2),
          "   Expected:", expect,
          "   correct?", is(ratsimp(val1 - expect) = 0)))$

/* ── C. (c) Leibniz rule with variable limits ────────── */
print("")$
print("════ C. (c) Leibniz rule: F'(x) = f(b(x))b'(x) - f(a(x))a'(x) ════")$

/* 1. One variable limit: d/dx ∫₀^(x²) e^(-t²) dt                */
print("")$
print("(1)  d/dx ∫₀^(x²) e^(-t²) dt")$
D1   : ratsimp(diff('integrate(exp(-t^2), t, 0, x^2), x))$
rule1: subst(x^2, t, exp(-t^2))*diff(x^2, x) - subst(0, t, exp(-t^2))*diff(0, x)$
print("     Maxima (diff applied to 'integrate):", D1)$
print("     Formula f(b(x))·b'(x) - f(a(x))·a'(x) :", ratsimp(rule1))$
print("     Difference (must be 0):", ratsimp(D1 - rule1))$
print("     [EXPECTED 2x·e^(-x⁴)]")$
/* The same with an explicit evaluation of the integral (erf) */
print("     With explicit evaluation of the integral:",
      ratsimp(diff(integrate(exp(-t^2), t, 0, x^2), x)))$

/* 2. TWO variable limits: d/dx ∫_x^(x²) sin(t²) dt              */
/*    Here the NEGATIVE term -f(a(x))·a'(x) shows up clearly.    */
print("")$
print("(2)  d/dx ∫ₓ^(x²) sin(t²) dt   — TWO variable limits")$
D2   : ratsimp(diff('integrate(sin(t^2), t, x, x^2), x))$
rule2: subst(x^2, t, sin(t^2))*diff(x^2, x) - subst(x, t, sin(t^2))*diff(x, x)$
print("     Maxima (diff applied to 'integrate):", D2)$
print("     Formula f(b(x))·b'(x) - f(a(x))·a'(x) :", ratsimp(rule2))$
print("     Difference (must be 0):", ratsimp(D2 - rule2))$
print("     [EXPECTED 2x·sin(x⁴) - sin(x²)]")$
print("     The term -sin(x²) comes from the LOWER limit a(x)=x")$
print("     (a'(x)=1) — without it the formula would be wrong.")$

/* Numerical check with finite differences at x = 1.3 */
print("")$
print("Numerical check at x = 1.3 (central finite difference):")$
block([x0, h, Iplus, Iminus, num, sym],
  x0 : 1.3,
  h  : 1.0e-4,
  Iplus  : quad_qags(sin(t^2), t, x0 + h, (x0 + h)^2)[1],
  Iminus : quad_qags(sin(t^2), t, x0 - h, (x0 - h)^2)[1],
  num : (Iplus - Iminus)/(2*h),
  sym : float(subst(x0, x, D2)),
  print("   finite differences :", num),
  print("   symbolic formula   :", sym),
  print("   |difference|       :", float(abs(num - sym))))$

/* ── D. (d) MVT for Integrals ────────────────────── */
print("")$
print("════ D. (d) MVT for Integrals: f(x)=x² on [0,3] ════")$

kill(c)$
a_mvt : 0$
b_mvt : 3$
I_mvt : integrate(x^2, x, a_mvt, b_mvt)$
print("∫₀³ x² dx =", I_mvt, "   [EXPECTED 9]")$

print("We look for c such that f(c)·(b-a) = ∫ₐᵇ f  ⇔  c²·3 = 9:")$
sols : solve(c^2*(b_mvt - a_mvt) = I_mvt, c)$
print("   solve →", sols)$
c_val : rhs(second(sols))$
print("   Admissible solution in [0,3]: c =", c_val, "=", float(c_val),
      "   [EXPECTED √3 ≈ 1.7320508]")$
print("   f(c) =", subst(c_val, c, c^2), "   height of the rectangle")$
print("   Check f(c)·(b-a) - ∫ₐᵇf =",
      ratsimp(subst(c_val, c, c^2)*(b_mvt - a_mvt) - I_mvt), "  (must be 0)")$
print("   Area of the rectangle =", float(subst(c_val, c, c^2)*(b_mvt - a_mvt)),
      "   Area under f =", float(I_mvt))$

/* Plot: f, the horizontal line at height f(c) = 3, and c */
h_rect : subst(c_val, c, c^2)$
plot2d([x^2, h_rect], [x, 0, 3], [y, 0, 9.5],
       [legend, "f(x) = x²", "y = f(c) = 3 (height of the rectangle)"],
       [xlabel, "x"], [ylabel, "y"],
       [title, "MVT for Integrals: equality of areas, c = sqrt(3)"])$

print("")$
print("End — Chapter 15.")$
