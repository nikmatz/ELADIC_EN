/* ── Part A: Definition and Matrix Operations ──────────── */
print("════ A. Matrix Operations ════");

/* A 3×3 matrix A */
A : matrix([1, 2, 1],
           [2, 1, 3],
           [1, 3, -1]);

print("Matrix A =");
print(A);

/* Determinant */
det_A : determinant(A);
print("det(A) =", det_A);

/* Rank */
r : rank(A);
print("rank(A) =", r);

/* ── Part B: Row Echelon Form & Null Space ─────────────── */
print("════ B. Row Echelon Form & Spaces ════");

/* Row echelon form — the pivots are immediately visible */
A_ech : echelon(A);
print("Echelon(A) =");
print(A_ech);

/* Null space: Ax = 0 */
ns : nullspace(A);
print("Null space of A:");
print(ns);

/* Column space */
cs : columnspace(A);
print("Column space of A:");
print(cs);

/* ── Part C: Linear Independence ───────────────────────── */
print("════ C. Linear Independence ════");

/* Vectors v1, v2, v3 */
v1 : matrix([1], [0], [1]);
v2 : matrix([0], [1], [1]);
v3 : matrix([1], [1], [2]);    /* v3 = v1 + v2  (linearly dependent) */

/* We check whether {v1, v2, v3} are linearly independent
   by forming a matrix with the v's as columns */
M : addcol(v1, v2, v3);
print("Matrix [v1|v2|v3] =");
print(M);
print("rank([v1|v2|v3]) =", rank(M));
print("→ If rank < 3  ⇒  linearly DEPENDENT");
print("  Here rank =", rank(M), "< 3, hence dependent ✓");

/* Check: v3 = v1 + v2 */
print("Check: v3 - v1 - v2 =", expand(v3 - v1 - v2));

/* ── Part D: The Gram-Schmidt Algorithm ────────────────── */
print("════ D. Gram-Schmidt Orthonormalization ════");

/* We start from linearly independent vectors */
u1 : matrix([1], [1], [0]);
u2 : matrix([1], [0], [1]);
u3 : matrix([0], [1], [1]);

/* Gram-Schmidt: e1, e2, e3 */
/* Step 1: e1 = u1 / |u1| */
e1_unnorm : u1;
e1 : e1_unnorm / sqrt(e1_unnorm . e1_unnorm);
e1 : ratsimp(e1);
print("e₁ =", e1);

/* Step 2: subtract the component parallel to e1 */
e2_unnorm : u2 - (u2 . e1) * e1;
e2_unnorm : ratsimp(e2_unnorm);
e2 : e2_unnorm / sqrt(e2_unnorm . e2_unnorm);
e2 : ratsimp(e2);
print("e₂ =", e2);

/* Step 3 */
e3_unnorm : u3 - (u3 . e1) * e1 - (u3 . e2) * e2;
e3_unnorm : ratsimp(e3_unnorm);
e3 : e3_unnorm / sqrt(e3_unnorm . e3_unnorm);
e3 : ratsimp(e3);
print("e₃ =", e3);

/* Check: e1·e2 = 0, e2·e3 = 0, e1·e3 = 0 */
print("Orthogonality:");
print("  e₁·e₂ =", ratsimp(e1 . e2));
print("  e₁·e₃ =", ratsimp(e1 . e3));
print("  e₂·e₃ =", ratsimp(e2 . e3));

/* Check that the norms are unit */
print("Norms:");
print("  |e₁|² =", ratsimp(e1 . e1));
print("  |e₂|² =", ratsimp(e2 . e2));
print("  |e₃|² =", ratsimp(e3 . e3));

/* ── Part E: Basis and Dimension ───────────────────────── */
print("════ E. Basis and Dimension ════");

/* We find a basis of the column space of A */
print("Basis of the column space of A:");
print(columnspace(A));
print("Dimension (rank) =", rank(A));

/* Nullity */
print("Nullity = n - rank = 3 -", rank(A), "=", 3 - rank(A));

print("════ End ════");
