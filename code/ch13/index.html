/* ============================================================
   maxima_aoristo.mac
   Chapter 13 — Indefinite Integral: Direct Integration & Substitution
   Matzakos, N. (2026). Elements of Linear Algebra, Differential & Integral Calculus. NewTech Publications.
   ============================================================

   COMMANDS USED:
     integrate(f, x)        → indefinite integral ∫f(x)dx (without the constant c)
     diff(F, x)             → differentiation — the safest check
     ratsimp(expr)          → simplification; ratsimp(diff(F,x)-f) = 0 ⇔ correct
     trigsimp(expr)         → simplification of trigonometric expressions
     'integrate(f, x)       → UNEVALUATED integral (quote)
     changevar(I, eq, u, x) → change of variable, step by step
     ev(expr, integrate)    → evaluation of the unevaluated integral
     solve(eq, c)           → solving for the constant c
     plot2d([...], [x,a,b]) → plot

   Book activity:
     (a) table of basic integrals + verification with diff
     (b) linearity
     (c) linear substitution: the factor 1/a
     (d) substitution u = g(x) with changevar
     (e) initial value problem y' = 3x²-4x+1, y(1) = 2

   Run with: wxMaxima (File → Batch File) or
                 maxima -b maxima_aoristo.mac
   ============================================================ */

/* ── A. (a) Table of basic integrals ────────────────── */
print("════ A. (a) Table of basic integrals ════")$

basics : [x^4, sqrt(x), exp(x), 1/x, sin(x), sec(x)^2,
          1/(1+x^2), 1/sqrt(1-x^2)]$

print("  f(x)  →  F(x) = ∫f(x)dx   (+ c)   and the check F'(x) - f(x)")$
for f in basics do
  block([F],
    F : integrate(f, x),
    print("   ", f, "  →  ", F,
          "      F'-f =", ratsimp(trigsimp(diff(F, x) - f))))$

/* All the checks together: the list must be all zeros */
print("")$
print("Checks F'-f (all must be 0):")$
print("   ", map(lambda([f], ratsimp(trigsimp(diff(integrate(f,x),x) - f))),
                 basics))$
print("  [Expected: ∫x⁴=x⁵/5, ∫√x=2x^(3/2)/3, ∫eˣ=eˣ, ∫dx/x=log x,")$
print("   ∫sin x=-cos x, ∫sec²x=tan x, ∫dx/(1+x²)=atan x, ∫dx/√(1-x²)=asin x]")$

/* ── B. (b) Linearity of the integral ───────────────── */
print("")$
print("════ B. (b) Linearity ════")$

g : 3*x^2 - 2*sin(x) + 1/x$
Ig : integrate(g, x)$
print("∫(3x² - 2·sin x + 1/x)dx =", Ig)$
print("  [Expected: x³ + 2·cos(x) + log(x)]")$

/* The same, term by term: 3∫x²dx - 2∫sin x dx + ∫dx/x */
Isum : 3*integrate(x^2, x) - 2*integrate(sin(x), x) + integrate(1/x, x)$
print("Sum of the parts: 3∫x²dx - 2∫sin x dx + ∫dx/x =", Isum)$
print("Difference of the two (must be 0):", ratsimp(Ig - Isum))$
print("Check with diff: F'-f =", ratsimp(diff(Ig, x) - g))$

/* ── C. (c) Linear substitution: the factor 1/a ──────── */
print("")$
print("════ C. (c) Linear substitution: ∫f(ax+b)dx = (1/a)F(ax+b) + c ════")$

linear_cases : [exp(2*x), cos(3*x), 1/(2*x+5), (4*x+1)^5]$
avals        : [2, 3, 2, 4]$

for k : 1 thru length(linear_cases) do
  block([f, F],
    f : linear_cases[k],
    F : integrate(f, x),
    print("   ∫", f, "dx =", F,
          "      a =", avals[k],
          "      F'-f =", ratsimp(diff(F, x) - f)))$

print("")$
print("  [Expected: ∫e^(2x)=e^(2x)/2, ∫cos3x=sin(3x)/3,")$
print("   ∫dx/(2x+5)=log(2x+5)/2, ∫(4x+1)⁵=(4x+1)⁶/24 (in expanded form)]")$
/* Maxima returns the expansion; it differs from (4x+1)⁶/24 by a constant */
print("(4x+1)⁶/24 - F(x) =", ratsimp((4*x+1)^6/24 - integrate((4*x+1)^5, x)),
      "  ⇒ the two forms differ by a constant (absorbed into c)")$

/* Symbolic confirmation of the rule: for an arbitrary f and constants a, b */
print("")$
print("Symbolic confirmation of the rule ∫f(ax+b)dx = (1/a)F(ax+b) + c:")$
kill(a, b)$
declare(a, constant, b, constant)$
for f in [exp(u), cos(u), 1/u, u^5] do
  block([F, cand],
    F    : integrate(f, u),                  /* F: the antiderivative of f */
    cand : subst(a*x + b, u, F)/a,           /* candidate antiderivative  */
    print("   f(u) =", f, "  →  d/dx[(1/a)F(ax+b)] - f(ax+b) =",
          ratsimp(diff(cand, x) - subst(a*x + b, u, f))))$
print("  [All four differences are 0 ⇒ the rule holds.]")$

/* ── D. (d) Substitution u = g(x) with changevar ──────── */
print("")$
print("════ D. (d) Substitution u = g(x) ════")$

/* 1. ∫2x·cos(x²)dx with u = x²  — step by step with changevar */
I1 : 'integrate(2*x*cos(x^2), x)$
print("Initial unevaluated form:", I1)$
J1 : changevar(I1, u = x^2, u, x)$
print("After u = x²:", J1)$
print("Result in u:", ev(J1, integrate))$
print("Back to x:", subst(x^2, u, ev(J1, integrate)),
      "   [Expected sin(x²)]")$
print("Directly:", integrate(2*x*cos(x^2), x),
      "   F'-f =", ratsimp(diff(integrate(2*x*cos(x^2),x),x) - 2*x*cos(x^2)))$

/* 2. ∫6x(3x²+1)⁵dx with u = 3x²+1 */
print("")$
I2 : 'integrate(6*x*(3*x^2+1)^5, x)$
J2 : changevar(I2, u = 3*x^2 + 1, u, x)$
print("∫6x(3x²+1)⁵dx: after u = 3x²+1 →", J2)$
print("   = ", integrate(6*x*(3*x^2+1)^5, x),
      "   [Expected (3x²+1)⁶/6]")$
print("   F'-f =", ratsimp(diff(integrate(6*x*(3*x^2+1)^5,x),x)
                           - 6*x*(3*x^2+1)^5))$

/* 3. ∫sin³x·cos x dx with u = sin x */
print("")$
S3 : integrate(sin(x)^3*cos(x), x)$
print("∫sin³x·cos x dx =", S3, "   [Expected sin⁴(x)/4]")$
print("   F'-f =", trigsimp(diff(S3, x) - sin(x)^3*cos(x)))$

/* 4. ∫(2x+3)/(x²+3x+5)dx with u = x²+3x+5  (numerator = u') */
print("")$
S4 : integrate((2*x+3)/(x^2+3*x+5), x)$
print("∫(2x+3)/(x²+3x+5)dx =", S4, "   [Expected log(x²+3x+5)]")$
print("   F'-f =", ratsimp(diff(S4, x) - (2*x+3)/(x^2+3*x+5)))$

/* ── E. (e) Initial value problem: y' = 3x²-4x+1, y(1) = 2 ── */
print("")$
print("════ E. (e) Initial value problem ════")$

kill(c)$
f_ivp : 3*x^2 - 4*x + 1$
F_ivp : integrate(f_ivp, x)$
print("y' =", f_ivp)$
print("General solution: y = F(x) + c  with F(x) =", F_ivp)$
print("  [Expected F(x) = x³ - 2x² + x]")$

print("Condition y(1) = 2:  F(1) + c = 2  with F(1) =", subst(1, x, F_ivp))$
sol_c : solve(subst(1, x, F_ivp) + c = 2, c)$
print("solve →", sol_c, "   [Expected c = 2]")$

c_val : rhs(first(sol_c))$
y_part : F_ivp + c_val$
print("Particular solution: y =", y_part)$
print("Check y(1) =", subst(1, x, y_part), "   (must be 2)")$
print("Check y'(x) - f(x) =", ratsimp(diff(y_part, x) - f_ivp),
      "   (must be 0)")$

/* The family y = F(x) + c for c = -2,-1,0,1,2 */
print("")$
print("Family of antiderivatives for c = -2,-1,0,1,2 — the same curve,")$
print("shifted vertically; the slope at x=1 is the same for every c:")$
for cc in [-2, -1, 0, 1, 2] do
  print("   c =", cc, "  y(2) =", subst(2, x, F_ivp + cc),
        "  y'(2) =", subst(2, x, diff(F_ivp + cc, x)),
        "  [f(2) =", subst(2, x, f_ivp), "]")$

plot2d([F_ivp - 2, F_ivp - 1, F_ivp, F_ivp + 1, F_ivp + 2],
       [x, -1, 3], [y, -6, 12],
       [legend, "c=-2", "c=-1", "c=0", "c=+1", "c=+2 (particular solution)"],
       [title, "y = x³-2x²+x + c  —  the particular solution y(1)=2 is c=+2"])$

print("")$
print("End — Chapter 13.")$
