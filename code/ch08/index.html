/* ============================================================
   maxima_synartiseis.mac
   Chapter 8 — Functions: Domain and Graphs
   Matzakos, N. (2026). Elements of Linear Algebra, Differential & Integral Calculus. NewTech Publications.
   ============================================================
   COMMANDS:
     define(f(x), expr)     -> definition of a function
     solve(eq, x)           -> solution of an EQUATION (endpoints/excluded points)
     fourier_elim([ineq],[x]) -> solution of an INEQUALITY (solve does not do inequalities)
     plot2d([f,g],[x,a,b])  -> common graph
     subst(-x, x, f)        -> substitution x -> -x (symmetries)
     charfun(condition)     -> 1 if it holds, otherwise 0 (piecewise)
     ratsimp(expr)          -> simplification of a rational expression
   ============================================================ */

load(fourier_elim)$   /* solve does NOT handle inequalities — we load a tool */

/* ═══ A. (a) DOMAIN ═════════════════════════════════════════ */
print("════ A. Domain ════")$

/* f(x) = sqrt(x-2)/(x^2-9):  x-2 >= 0  AND  x^2-9 # 0            */
define(f(x), sqrt(x-2)/(x^2-9))$
print("f(x) =", f(x))$
print("  Radicand:  x-2 >= 0  ->", fourier_elim([x-2 >= 0], [x]),
      "  that is x >= 2")$
print("  Denominator = 0 at:", solve(x^2-9 = 0, x), " (x = -3, x = 3)")$
print("  DOMAIN of f: [2,3) ∪ (3,+∞)   (-3 is already excluded by x>=2) ✓")$
print("  Check: f(2) =", f(2), " (= 0 ✓),  f(6) =", f(6), " = ",
      float(f(6)), " (= 2/27 = 0.074074 ✓)")$

/* g(x) = ln(4-x^2):  4-x^2 > 0                                   */
define(g(x), log(4-x^2))$
print("g(x) =", g(x))$
print("  Requirement 4-x^2 > 0  ->", fourier_elim([4-x^2 > 0], [x]))$
print("  Endpoints from solve(4-x^2=0,x):", solve(4-x^2 = 0, x))$
print("  DOMAIN of g: (-2,2) ✓")$
print("  Check: g(0) =", g(0), " (= log(4) = ", float(g(0)), " ✓)")$

/* h(x) = 1/sqrt(1-x^2):  1-x^2 > 0 (strictly, not >=)            */
define(h(x), 1/sqrt(1-x^2))$
print("h(x) =", h(x))$
print("  Requirement 1-x^2 > 0  ->", fourier_elim([1-x^2 > 0], [x]))$
print("  Endpoints from solve(1-x^2=0,x):", solve(1-x^2 = 0, x))$
print("  DOMAIN of h: (-1,1) ✓")$
print("  Check: h(0) =", h(0), " (= 1 ✓)")$

/* Graphical confirmation — the intervals are ALWAYS given in       */
/* increasing order and we avoid the zeros of the denominator       */
plot2d(f(x), [x, 2, 2.9], [y, -3, 1],
       [xlabel, "x"], [ylabel, "f(x)"],
       [title, "f(x)=sqrt(x-2)/(x^2-9) on [2,3)"])$
plot2d(f(x), [x, 3.1, 8], [y, 0, 3],
       [xlabel, "x"], [ylabel, "f(x)"],
       [title, "f(x)=sqrt(x-2)/(x^2-9) on (3,+oo)"])$
plot2d(g(x), [x, -1.98, 1.98], [y, -5, 2],
       [xlabel, "x"], [ylabel, "g(x)"],
       [title, "g(x)=ln(4-x^2), domain (-2,2)"])$
plot2d(h(x), [x, -0.98, 0.98], [y, 0, 6],
       [xlabel, "x"], [ylabel, "h(x)"],
       [title, "h(x)=1/sqrt(1-x^2), domain (-1,1)"])$

/* ═══ B. (b) SYMMETRIES AND COMPOSITIONS ═══════════════════ */
print("════ B. Symmetries and operations ════")$

/* Even: f(-x) - f(x) = 0.  Odd: f(-x) + f(x) = 0.                */
u1 : x^4 - 3*x^2$
u2 : x^3 - x$
u3 : exp(x) + exp(-x)$
for e in [u1, u2, u3] do block([em, ep],
    em : ratsimp(subst(-x, x, e) - e),
    ep : ratsimp(subst(-x, x, e) + e),
    print("  ", e),
    print("     f(-x)-f(x) =", em, " | f(-x)+f(x) =", ep),
    if em = 0 then print("     -> EVEN (symmetric about the y-axis)")
    elseif ep = 0 then print("     -> ODD (symmetric about the origin)")
    else print("     -> neither even nor odd") )$
print("  Expected: x^4-3x^2 EVEN ✓, x^3-x ODD ✓, e^x+e^-x EVEN ✓")$

plot2d([u1, u2], [x, -2, 2], [y, -3, 3],
       [legend, "x^4-3x^2 (even)", "x^3-x (odd)"],
       [xlabel, "x"], [title, "Symmetries"])$

/* Compositions with f(x)=x^2+1 and g(x)=sqrt(x-1)                */
define(F(x), x^2 + 1)$
define(G(x), sqrt(x-1))$
print("F(x) =", F(x), ",  G(x) =", G(x))$
print("  (F o G)(x) = F(G(x)) =", ratsimp(F(G(x))), "  (= x ✓)")$
print("  (G o F)(x) = G(F(x)) =", ratsimp(G(F(x))), "  (= |x| ✓ — Maxima simplifies it to abs(x))")$
print("  Numerically at x = -3:")$
print("     F(G(-3)) : G(-3)=sqrt(-4) is NOT defined in R (domain of G: x>=1)")$
print("     G(F(-3)) =", G(F(-3)), " (= 3 = |-3| ✓, NOT -3)")$
print("  WHY THEY DIFFER: F o G is defined only for x>=1 and gives x,")$
print("  while G o F is defined for every x and gives |x|. Composition is not")$
print("  commutative and its domain changes with the order.")$

plot2d([x, sqrt(x^2)], [x, -3, 3], [y, -3, 3],
       [legend, "(FoG)(x) = x  (only for x>=1)", "(GoF)(x) = |x|"],
       [xlabel, "x"], [title, "F o G  versus  G o F"])$

/* ═══ C. (c) ONE-TO-ONE AND INVERSE ════════════════════════ */
print("════ C. f(x)=(2x+1)/(x-3): one-to-one and its inverse ════")$

define(q(x), (2*x+1)/(x-3))$
print("q(x) =", q(x), "  (domain: R \\ {3})")$

/* One-to-one: q(a)=q(b) must imply a=b.                          */
print("  solve(q(a)=q(b), a) =", solve(q(aa) = q(bb), aa),
      "  ->  a = b, hence it IS one-to-one ✓")$
print("  Alternatively q'(x) =", ratsimp(diff(q(x), x)),
      " = -7/(x-3)^2 < 0 everywhere -> strictly decreasing on each branch ✓")$

/* Inverse: we solve y = q(x) for x.                              */
solq : solve(y = q(x), x)$
print("  solve(y = q(x), x) =", solq)$
define(qinv(x), subst(x, y, rhs(solq[1])))$
print("  q^(-1)(x) =", qinv(x), "  (= (3x+1)/(x-2) ✓)")$
print("  Check q(q^(-1)(x)) =", ratsimp(q(qinv(x))), "  (= x ✓)")$
print("  Check q^(-1)(q(x)) =", ratsimp(qinv(q(x))), "  (= x ✓)")$
print("  Numerically: q(5) =", q(5), " (= 11/2) and q^(-1)(11/2) =",
      qinv(11/2), " (= 5 ✓)")$

/* Common plot of q, q^(-1), y = x : symmetry about y = x.        */
plot2d([q(x), qinv(x), x], [x, -7, 10], [y, -7, 10],
       [legend, "q(x)=(2x+1)/(x-3)", "q^(-1)(x)=(3x+1)/(x-2)", "y = x"],
       [xlabel, "x"], [ylabel, "y"],
       [title, "Symmetry of q and of its inverse about y=x"])$
print("  Remark: the vertical asymptote x=3 of q becomes the horizontal")$
print("  asymptote y=3 of q^(-1) — exactly the mirror symmetry about y=x.")$

/* ═══ D. (d) TRANSFORMATIONS AND PIECEWISE FUNCTIONS ═══════ */
print("════ D. Transformations of f(x)=x^2 and the piecewise p ════")$

define(s(x), x^2)$
print("Base: s(x) = x^2")$
print("  s(x-2) =", expand(s(x-2)),
      "  -> HORIZONTAL shift 2 units to the RIGHT (vertex at x=2)")$
print("  s(x)+3 =", s(x)+3,
      "  -> VERTICAL shift 3 units UP (vertex at (0,3))")$
print("  2*s(x) =", 2*s(x),
      "  -> VERTICAL stretch by 2 (narrower parabola)")$
print("  s(2x)  =", expand(s(2*x)),
      "  -> HORIZONTAL compression by a factor 1/2 (also narrower)")$
print("  -s(x)  =", -s(x),
      "  -> REFLECTION in the x-axis (opens downward)")$
print("  Check at x=3: s(3)=", s(3), "(=9), s(3-2)=", s(3-2), "(=1),")$
print("     s(3)+3=", s(3)+3, "(=12), 2*s(3)=", 2*s(3), "(=18), s(2*3)=",
      s(2*3), "(=36), -s(3)=", -s(3), "(=-9) ✓")$

plot2d([s(x), s(x-2), s(x)+3, 2*s(x), s(2*x), -s(x)], [x, -4, 5], [y, -6, 14],
       [legend, "f(x)=x^2", "f(x-2)", "f(x)+3", "2f(x)", "f(2x)", "-f(x)"],
       [xlabel, "x"], [ylabel, "y"],
       [title, "Transformations of f(x)=x^2"])$

/* Piecewise p(x) = x^2 for x<1,  2x-1 for x>=1, with charfun.    */
p(x) := x^2*charfun(x < 1) + (2*x-1)*charfun(x >= 1)$
print("p(x) = x^2*charfun(x<1) + (2x-1)*charfun(x>=1)")$
for a in [-1, 0, 0.5, 0.9, 1, 1.5, 3] do
    print("   p(", a, ") =", p(a))$

/* Continuity at x=1: comparison of the one-sided limits with p(1). */
lm : limit(x^2, x, 1, minus)$
lp : limit(2*x-1, x, 1, plus)$
print("  lim x->1^-  x^2   =", lm, " (= 1 ✓)")$
print("  lim x->1^+  2x-1  =", lp, " (= 1 ✓)")$
print("  p(1) =", p(1), " (= 1 ✓)")$
if lm = lp and lp = p(1) then
   print("  -> The two one-sided limits and the value agree: p IS CONTINUOUS at x=1")
else
   print("  -> p is NOT continuous at x=1")$

plot2d(p(x), [x, -2, 3], [y, -1, 6],
       [xlabel, "x"], [ylabel, "p(x)"],
       [title, "p(x): x^2 for x<1, 2x-1 for x>=1 — no jump at x=1"])$

print("── End of maxima_synartiseis.mac ──")$
