/* ============================================================
   maxima_eigen.mac
   Chapter 5 — Eigenvalues, Characteristic Polynomial, Eigenspaces
   Matzakos, N. (2026). Elements of Linear Algebra, Differential & Integral Calculus. NewTech Publications.
   ============================================================

   COMMANDS USED:
     load("eigen")           → load the eigenvalue package
     charpoly(A, lam)        → characteristic polynomial det(A - lam·I)
     solve(p = 0, lam)       → roots of the characteristic polynomial
     eigenvalues(A)          → [[eigenvalues], [multiplicities]]
     eigenvectors(A)         → eigenvalues + multiplicities + eigenvectors
     nullspace(A - lam*I)    → eigenspace (kernel) of the eigenvalue lam
     ident(n), rank(M)       → identity matrix, rank

   Book activity: (a) characteristic polynomial & tr/det,
     (b) the eigenspaces of A, (c) eigenvalues of the triangular C,
     (d) algebraic vs geometric multiplicity.

   Run with: wxMaxima (Load/File) or
                 maxima -b maxima_eigen.mac
   ============================================================ */

load("eigen")$

/* Helper: equality of matrices through their difference */
mat_equal(M, N) := is( ratsimp(M - N) = zeromatrix(length(M), length(M[1])) )$

/* Helper: list of the eigenvalues WITH their multiplicities.
   eigenvalues() returns [[lam1,lam2,...],[m1,m2,...]] —
   solve() would give only the DISTINCT roots, so the sums
   and the products would be wrong for repeated eigenvalues. */
eigs_with_mult(M) := block([ev, vals, mults, out],
    ev : eigenvalues(M),
    vals : ev[1], mults : ev[2], out : [],
    for i : 1 thru length(vals) do
        for k : 1 thru mults[i] do out : endcons(vals[i], out),
    out)$

/* ── A. (a) Characteristic polynomial, tr(A)=sum, det(A)=prod ─ */
print("════ A. (a) Characteristic Polynomial ════");

A : matrix([4, 1],
           [2, 3]);
print("A ="); print(A);

p_A : expand(charpoly(A, lam));
print("p(lam) = det(A - lam·I) =", p_A);
print("  in factored form:", factor(p_A));
print("  [Expected: (lam-5)(lam-2) ✓]");

print("Roots: solve(p(lam)=0, lam) →", solve(p_A = 0, lam));

/* Eigenvalues WITH multiplicities (needed for tr/det) */
print("eigenvalues(A) =", eigenvalues(A));
evA : eigs_with_mult(A);
print("Eigenvalues with multiplicities: lam =", evA, "  [Expected [2,5]]");

tr_A : A[1][1] + A[2][2];
sum_A : apply("+", evA);
det_A : determinant(A);
prod_A : apply("*", evA);
print("tr(A) =", tr_A, " | sum(lam_i) =", sum_A, " | Equality:", is(tr_A = sum_A));
print("det(A) =", det_A, " | prod(lam_i) =", prod_A, " | Equality:", is(det_A = prod_A));

/* Check that det(A - lam_i·I) = 0 for every eigenvalue */
for lv in evA do
    print("  lam =", lv, "→ det(A - lam·I) =", ratsimp(determinant(A - lv*ident(2))));

/* ── B. (b) Eigenspaces of A and check A·v = lam·v ──────── */
print("════ B. (b) Eigenspaces — nullspace(A - lam·I) ════");

for lv in listify(setify(evA)) do block([Mk, ns, basis, v],
    Mk : A - lv*ident(2),
    print("─ lam =", lv, ": A - lam·I ="), print(Mk),
    ns : nullspace(Mk),
    print("  nullspace(A - lam·I) =", ns),
    basis : args(ns),
    print("  dimension of the eigenspace (geometric multiplicity) =", length(basis)),
    for v in basis do (
        print("  v =", transpose(v), "ᵀ"),
        print("  A·v =", transpose(ratsimp(A . v)), "ᵀ",
              " | lam·v =", transpose(ratsimp(lv * v)), "ᵀ"),
        print("  A·v = lam·v ;  Check:", mat_equal(A . v, lv * v))
    )
);

print("eigenvectors(A) =", eigenvectors(A));

/* ── C. (c) Lower triangular C — why lam = the diagonal entries ─ */
print("════ C. (c) Triangular matrix C ════");

C : matrix([2, 0, 0],
           [1, 3, 0],
           [0, 1, 2]);
print("C ="); print(C);

print("C - lam·I ="); print(C - lam*ident(3));
p_C : charpoly(C, lam);
print("p(lam) = det(C - lam·I) =", factor(expand(p_C)));
print("  [Expected: (lam-2)²(lam-3) ✓]");
print("Why? The determinant of a TRIANGULAR matrix is the product of");
print("its diagonal entries. C - lam·I is still lower triangular, so");
print("det(C - lam·I) = (2-lam)(3-lam)(2-lam) → the roots are EXACTLY 2, 3, 2.");
print("Product of the diagonal entries of C - lam·I:",
      factor(expand((C[1][1]-lam)*(C[2][2]-lam)*(C[3][3]-lam))));
print("Identical with p(lam) ;  Check:",
      is(ratsimp(expand(p_C) - expand((C[1][1]-lam)*(C[2][2]-lam)*(C[3][3]-lam))) = 0));

print("eigenvectors(C) ="); print(eigenvectors(C));

evC : eigs_with_mult(C);
print("Eigenvalues with multiplicities:", evC, "  [Expected [2,2,3]]");
tr_C : C[1][1] + C[2][2] + C[3][3];
print("tr(C) =", tr_C, " | sum(lam_i) =", apply("+", evC),
      " | Equality:", is(tr_C = apply("+", evC)));
print("det(C) =", determinant(C), " | prod(lam_i) =", apply("*", evC),
      " | Equality:", is(determinant(C) = apply("*", evC)));

/* ── D. (d) Algebraic vs geometric multiplicity of C ────── */
print("════ D. (d) Algebraic vs Geometric Multiplicity ════");

evsC : eigenvalues(C)$
valsC : evsC[1]$
multC : evsC[2]$

defective : false$
for i : 1 thru length(valsC) do block([lv, ma, Mk, ns, mg],
    lv : valsC[i],
    ma : multC[i],
    Mk : C - lv*ident(3),
    ns : nullspace(Mk),
    mg : length(args(ns)),
    print("─ lam =", lv),
    print("  algebraic multiplicity m_a =", ma),
    print("  nullspace(C - lam·I) =", ns),
    print("  geometric multiplicity m_g = dim ker = n - rank =",
          3 - rank(Mk), " = ", mg),
    if mg < ma then (
        defective : true,
        print("  m_g < m_a  →  DEFECTIVE eigenvalue")
    ) else
        print("  m_g = m_a  →  regular eigenvalue")
);

print("Sum of the geometric multiplicities =",
      sum(3 - rank(C - valsC[i]*ident(3)), i, 1, length(valsC)),
      "( < 3 → C is NOT diagonalizable )");
print("There is a defective eigenvalue:", defective);

print("════ End ════");
