/* ── Part A: Basic Operations with Complex Numbers ──────── */
print("════ A. Basic Operations ════");

z1 : 3 + 4*%i;          /* z₁ = 3 + 4i */
z2 : 1 - 2*%i;          /* z₂ = 1 - 2i */

print("z₁ =", z1, "   z₂ =", z2);
print("Addition:        z₁ + z₂ =", expand(z1 + z2));
print("Subtraction:     z₁ - z₂ =", expand(z1 - z2));
print("Multiplication:  z₁·z₂  =", expand(z1 * z2));
print("Division:        z₁/z₂  =", rectform(z1 / z2));

/* Modulus and Argument */
print("");
print("Modulus  |z₁| =", abs(z1),    " (= √(3²+4²) = 5)");
print("Modulus  |z₂| =", abs(z2));
print("Argument arg(z₁) =", carg(z1), "rad =",
      float(carg(z1)) * 180 / %pi, "°");

/* ── Part B: Exponential Form — AC Circuit Phasors ─────── */
print("════ B. Exponential Form — Phasors ════");

/* Two phasors of an electrical circuit */
z_A : 5 * exp(%i * %pi / 3);   /* |z_A| = 5, angle = 60° */
z_B : 2 * exp(%i * %pi / 6);   /* |z_B| = 2, angle = 30° */

print("Phasor A: z_A = 5·e^(i·pi/3)  →  rectangular:", rectform(z_A));
print("Phasor B: z_B = 2·e^(i·pi/6)  →  rectangular:", rectform(z_B));

print("");
print("(b1) Product z_A·z_B   = ", simplify(z_A * z_B));
print("     Rectangular:      = ", rectform(simplify(z_A * z_B)));
print("     [Modulus = 5·2 = 10, Angle = 60°+30° = 90°  →  10i ✓]");

print("");
print("(b2) Quotient z_A/z_B  = ", simplify(z_A / z_B));
print("     Rectangular:      = ", rectform(simplify(z_A / z_B)));
print("     [Modulus = 5/2 = 2.5, Angle = 60°−30° = 30°]");

/* ── Part C: De Moivre's Theorem ────────────────────────── */
print("════ C. De Moivre's Theorem ════");
/* (cos t + i sin t)^n = cos(nt) + i sin(nt) */

theta : %pi / 3;   /* 60° */
n     : 3;

lhs : expand((cos(theta) + %i * sin(theta))^n);
rhs : cos(n * theta) + %i * sin(n * theta);

print("(cos 60° + i sin 60°)³ =", float(lhs));
print("cos 180° + i sin 180°  =", float(rhs));
print("Equality:", is(float(lhs) = float(rhs)));   /* true */

/* Application: (5e^(i·pi/3))³ */
print("");
z_cubed : simplify((5 * exp(%i * %pi / 3))^3);
print("(5·e^(i·pi/3))³ =", z_cubed, "=", rectform(z_cubed));
print("[Expected: modulus = 5³=125, angle = 3·60°=180°  →  −125 ✓]");

/* ── Part D: Roots of Complex Numbers ───────────────────── */
print("════ D. nth Roots of Complex Numbers ════");
/* z^n = w  has exactly n roots:
   z_k = r^(1/n) · e^(i(t+2k·pi)/n),  k = 0,1,...,n-1  */

/* Square roots of −1 */
print("Roots of x² = -1:");
sol : solve(x^2 = -1, x);
print("  x =", sol);

/* Cube roots of 8 */
print("Roots of x³ = 8:");
sol3 : solve(x^3 = 8, x);
print("  x =", sol3);
/* Display numerically */
print("  Numerically:", float(sol3));

/* ── Part E: Division of Complex Numbers — Verification ── */
print("════ E. Useful Identities ════");

z : a + %i * b;   /* a general complex number */
conj_z : a - %i * b;

print("z·conj(z) =", expand(z * conj_z), " (= a² + b² = |z|²  ✓)");
print("z + conj(z) =", expand(z + conj_z), " (= 2a = 2·Re(z)  ✓)");
print("z - conj(z) =", expand(z - conj_z), " (= 2bi = 2i·Im(z) ✓)");

print("════ End ════");
