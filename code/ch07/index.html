     eigenvectors(P)    -> eigenvalues together with eigenvectors
     A^^k . x0          -> power of a matrix (^^) times a vector
     invert(M)          -> inverse matrix
     float(expr)        -> decimal form
   ------------------------------------------------------------
   We work with RATIONAL numbers (8/10 instead of 0.8) so that the
   eigenvalues come out exact; we print with float() where needed.
   ============================================================ */

/* ── A. (a) Evolution x_{k+1} = A x_k, eigenvalues, stability ─ */
print("════ A. Evolution of the system x_{k+1} = A x_k ════")$

A  : matrix([8/10, 0], [0, 6/10])$
x0 : matrix([1], [1])$
print("A  =", A)$
print("x0 =", transpose(x0))$

x1  : A . x0$
x2  : A . x1$
x10 : A^^10 . x0$
print("x1  =", transpose(x1),  " = (0.8 ; 0.6)  (= [4/5, 3/5] ✓)")$
print("x2  =", transpose(x2),  " = (0.64 ; 0.36) (= [16/25, 9/25] ✓)")$
print("x10 =", float(transpose(x10)),
      " = (0.8^10 ; 0.6^10) = (0.1073741824 ; 0.0060466176) ✓")$

evA : eigenvalues(A)$
print("Eigenvalues of A:", evA[1], " (= [4/5, 3/5] ✓)")$
rhoA : lmax(map(abs, evA[1]))$
print("Spectral radius rho(A) =", rhoA, "=", float(rhoA), " (= 0.8 ✓)")$

/* The conclusion is NOT hard-coded: it follows from a comparison. */
if rhoA < 1 then
   print("rho(A) < 1  =>  x_k -> 0 : asymptotically STABLE")
elseif rhoA = 1 then
   print("rho(A) = 1  =>  borderline case: bounded orbits, not -> 0")
else
   print("rho(A) > 1  =>  UNSTABLE: the orbits move away")$

/* ── B. (b) The system with B is unstable, through rho(B) ──── */
print("════ B. Instability for the matrix B ════")$

B : matrix([11/10, 2/10], [1/10, 9/10])$
print("B =", B)$
evB : eigenvalues(B)$
print("Eigenvalues of B:", evB[1])$
print("  numerically:", float(evB[1]),
      " (= 1 - sqrt(3)/10 = 0.82679... and 1 + sqrt(3)/10 = 1.17320... ✓)")$
rhoB : lmax(map(abs, float(evB[1])))$
print("rho(B) =", rhoB, " (= 1.1732050807568877 ✓)")$

if rhoB < 1 then
   print("rho(B) < 1  =>  stable")
elseif rhoB = 1 then
   print("rho(B) = 1  =>  borderline case")
else
   print("rho(B) > 1  =>  UNSTABLE ✓ (as the question asks)")$

/* Confirmation: the orbit of B moves away (the terms grow). */
y0 : matrix([1], [1])$
for m : 5 thru 20 step 5 do
    print("  B^", m, ". y0 =", float(transpose(B^^m . y0)))$

/* ── C. (c) Markov chain: stationary distribution pi ───────── */
print("════ C. Stationary distribution of a Markov chain ════")$

Pm : matrix([9/10, 2/10], [1/10, 8/10])$
print("P =", Pm, "  (columns summing to 1)")$

[valsP, vecsP] : eigenvectors(Pm)$
print("Eigenvalues of P:", valsP[1], " (= [7/10, 1] ✓)")$

/* NOTE: we do not assume the position of lam=1 — we SEARCH for it. */
idx : 0$
for i : 1 thru length(valsP[1]) do
    if valsP[1][i] = 1 then idx : i$
print("The eigenvalue lam=1 is found at position:", idx)$

v : transpose(matrix(vecsP[idx][1]))$
print("Eigenvector for lam=1 (before normalization):", transpose(v))$

s : v[1,1] + v[2,1]$
print("Sum of the components:", s, " (must be ≠ 0)")$
pist : v / s$
print("Stationary distribution pi =", transpose(pist), " (= [2/3, 1/3] ✓)")$
print("  numerically:", float(transpose(pist)), " (= [0.6667 ; 0.3333] ✓)")$
print("Check P.pi = pi :", is(ratsimp(Pm . pist - pist) = matrix([0],[0])),
      "  (true ✓)")$
print("Check that the sum = 1 :", is(pist[1,1] + pist[2,1] = 1), "  (true ✓)")$

/* ── D. (d) Closed form for A^k through diagonalization ───── */
print("════ D. A^k in closed form (diagonalization) ════")$

kill(k)$   /* k must be a free symbol */

[valsA, vecsA] : eigenvectors(A)$
/* We collect ALL the eigenvectors as columns of S.            */
lam  : []$
cols : []$
for i : 1 thru length(valsA[1]) do
    for j : 1 thru valsA[2][i] do (
        lam  : append(lam,  [valsA[1][i]]),
        cols : append(cols, [vecsA[i][j]]) )$
S  : transpose(apply(matrix, cols))$
Dm : apply(matrix, makelist(makelist(if i = j then lam[i] else 0,
                                     j, 1, length(lam)),
                            i, 1, length(lam)))$
print("S =", S, "   D =", Dm)$
print("Check A = S.D.S^(-1):",
      is(ratsimp(S . Dm . invert(S) - A) = matrix([0,0],[0,0])), " (true ✓)")$

Dk : apply(matrix, makelist(makelist(if i = j then lam[i]^k else 0,
                                     j, 1, length(lam)),
                            i, 1, length(lam)))$
Ak : ratsimp(S . Dk . invert(S))$
print("A^k =", Ak, "   that is A^k = diag((4/5)^k, (3/5)^k) ✓")$

/* Verification of the closed form for k = 10.                 */
print("A^10 from the formula:", float(subst(10, k, Ak)))$
print("A^10 directly     :", float(A^^10), "  (they agree ✓)")$
print("x_k = A^k x0 =", transpose(ratsimp(Ak . x0)))$

/* ── Graphical insight: the components decay geometrically ── */
kill(t)$
plot2d([(4/5)^t, (3/5)^t], [t, 0, 20],
       [legend, "x1(k) = 0.8^k", "x2(k) = 0.6^k"],
       [xlabel, "k"], [ylabel, "component"],
       [title, "Asymptotic stability: rho(A) = 0.8 < 1"])$

print("── End of maxima_dynamics.mac ──")$
