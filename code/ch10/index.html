/* ============================================================
   maxima_paragogos.mac
   Κεφάλαιο 10 — Παράγωγος: Κανόνες, Ανώτερης Τάξης, Εφαπτόμενη
   Ματζάκος, Ν. (2026). Στοιχεία Γραμμικής Άλγεβρας, Διαφορικού & Ολοκληρωτικού Λογισμού. NewTech Publications.
   ============================================================

   ΕΝΤΟΛΕΣ ΠΟΥ ΧΡΗΣΙΜΟΠΟΙΟΥΝΤΑΙ:
     diff(f, x)                     -> f'(x)
     diff(f, x, n)                  -> n-οστή παράγωγος
     at(diff(f,x), x=a)             -> τιμή της παραγώγου στο a
     subst(a, x, expr)              -> αντικατάσταση x -> a
     trigsimp(...) , ratsimp(...)   -> απλοποίηση
     limit((f(x+h)-f(x))/h, h, 0)   -> παράγωγος από τον ΟΡΙΣΜΟ

   Εκτελέστε με: wxMaxima (File/Batch) ή
                 maxima -b maxima_paragogos.mac
   ============================================================ */

/* ── Α. Βασικοί κανόνες παραγώγισης ─────────────────────── */
print("════ Α. Βασικοί κανόνες ════")$

fA1 : 3*x^4 - 5*x^2 + 2*x - 7$
fA2 : sin(x)$
fA3 : exp(3*x)$
fA4 : log(x^2 + 1)$

print("(3x^4-5x^2+2x-7)' =", diff(fA1, x), "   (αναμ. 12x^3-10x+2)")$
print("(sin x)'          =", diff(fA2, x), "   (αναμ. cos x)")$
print("(e^(3x))'         =", diff(fA3, x), "   (αναμ. 3e^(3x))")$
print("(ln(x^2+1))'      =", diff(fA4, x), "   (αναμ. 2x/(x^2+1))")$

/* Επαλήθευση με χαρτί και μολύβι: δύναμη, αλυσίδα, λογάριθμος. */
print("Έλεγχος (διαφορά από το αναμενόμενο, πρέπει 0):",
      ratsimp(diff(fA1,x) - (12*x^3-10*x+2)),
      ratsimp(diff(fA2,x) - cos(x)),
      ratsimp(diff(fA3,x) - 3*exp(3*x)),
      ratsimp(diff(fA4,x) - 2*x/(x^2+1)))$

/* ── Β. Γινόμενο, πηλίκο, αλυσίδα ───────────────────────── */
print("")$
print("════ Β. Γινόμενο — Πηλίκο — Αλυσίδα ════")$

/* Κανόνας γινομένου: (uv)' = u'v + uv' */
gB : x^2*sin(x)$
print("(x^2 sin x)' =", diff(gB, x), "   (αναμ. 2x sin x + x^2 cos x)")$

/* Κανόνας πηλίκου: (u/v)' = (u'v - uv')/v^2 */
hB : (x^3 + 1)/(x^2 - 1)$
print("((x^3+1)/(x^2-1))' =", factor(ratsimp(diff(hB, x))),
      "   (αναμ. x(x-2)/(x-1)^2)")$

/* Κανόνας αλυσίδας: [f(g(x))]' = f'(g(x)) g'(x) */
kB : sin(x^2 + 1)$
print("[sin(x^2+1)]' =", diff(kB, x), "   (αναμ. 2x cos(x^2+1))")$

print("Έλεγχοι (πρέπει 0,0,0):",
      ratsimp(diff(gB,x) - (2*x*sin(x) + x^2*cos(x))),
      ratsimp(diff(hB,x) - x*(x-2)/(x-1)^2),
      ratsimp(diff(kB,x) - 2*x*cos(x^2+1)))$

/* ── Γ. Ανώτερης τάξης: f(x) = e^x cos x ────────────────── */
print("")$
print("════ Γ. Παράγωγοι ανώτερης τάξης: f(x)=e^x cos(x) ════")$

fC : exp(x)*cos(x)$
for n:1 thru 4 do
    print("  f^(", n, ")(x) =", expand(diff(fC, x, n)))$

/* Η ζητούμενη ταυτότητα: f^(4) = -4 f  */
print("  f^(4) + 4 f =", expand(diff(fC, x, 4) + 4*fC),
      "  =>  f^(4) = -4f  (επαληθεύεται)")$

/* Ανώτερης τάξης και για το sin: κύκλος περιόδου 4 */
print("  Για σύγκριση, sin^(n)(x):")$
for n:1 thru 4 do print("    n =", n, "->", diff(sin(x), x, n))$

/* ── Δ. Εφαπτόμενη ευθεία: f(x)=x^3-2x+1 στο x0=1 ───────── */
print("")$
print("════ Δ. Εφαπτόμενη της f(x)=x^3-2x+1 στο x0=1 ════")$

fD : x^3 - 2*x + 1$
x0 : 1$
y0 : subst(x0, x, fD)$
m  : at(diff(fD, x), x = x0)$          /* at(): τιμή παραγώγου στο x0 */
tang : expand(y0 + m*(x - x0))$

print("  f(1) =", y0, "   f'(x) =", diff(fD,x), "   f'(1) =", m)$
print("  Εφαπτόμενη: y =", tang, "   (αναμ. y = x - 1)")$
print("  Έλεγχος (πρέπει 0):", ratsimp(tang - (x - 1)))$

/* Γράφημα f, f', f'' και της εφαπτομένης στο [-2, 2.5] */
plot2d([fD, diff(fD,x), diff(fD,x,2), tang], [x, -2, 2.5],
       [legend, "f", "f'", "f''", "εφαπτόμενη στο x0=1"],
       [title, "f(x)=x^3-2x+1 : f, f', f'' και εφαπτόμενη"])$

/* ── Ε. Η παράγωγος από τον ΟΡΙΣΜΟ (όριο) ──────────────── */
print("")$
print("════ Ε. Ορισμός: f'(x) = lim_{h->0} (f(x+h)-f(x))/h ════")$

/* f(x) = x^2 */
print("  x^2 : λόγος μεταβολής =", ratsimp(((x+h)^2 - x^2)/h))$
print("        όριο h->0        =", limit(((x+h)^2 - x^2)/h, h, 0),
      "   (diff: ", diff(x^2, x), ")")$

/* f(x) = sqrt(x) */
print("  sqrt(x) : όριο h->0    =", limit((sqrt(x+h) - sqrt(x))/h, h, 0),
      "   (diff: ", diff(sqrt(x), x), ")")$

/* Ένα ακόμη: f(x)=sin x — ο ορισμός δίνει cos x */
print("  sin(x)  : όριο h->0    =", limit((sin(x+h) - sin(x))/h, h, 0))$

/* Πλευρικά όρια: η |x| ΔΕΝ είναι παραγωγίσιμη στο 0 */
print("")$
print("  |x| στο 0 — από δεξιά   :", limit((abs(0+h) - abs(0))/h, h, 0, plus))$
print("  |x| στο 0 — από αριστερά:", limit((abs(0+h) - abs(0))/h, h, 0, minus))$
print("  Τα πλευρικά όρια διαφέρουν => δεν υπάρχει παράγωγος στο 0.")$

/* ── ΣΤ. Άρρητη παραγώγιση: x^2 + y^2 = 25 ──────────────── */
print("")$
print("════ ΣΤ. Άρρητη παραγώγιση: x^2 + y^2 = 25 ════")$

/* Θεώρημα πεπλεγμένης: dy/dx = -F_x / F_y  (μερικές παράγωγοι).
   ΠΡΟΣΟΧΗ: χωρίς depends(y,x) η diff είναι μερική — αυτό θέλουμε. */
eq : x^2 + y^2 - 25$
dy_dx : ratsimp(-diff(eq, x)/diff(eq, y))$
print("  F_x =", diff(eq,x), "   F_y =", diff(eq,y))$
print("  dy/dx =", dy_dx, "   (αναμ. -x/y)")$
print("  Έλεγχος (πρέπει 0):", ratsimp(dy_dx + x/y))$

/* Εναλλακτικά, με depends(y,x) λύνουμε ως προς 'diff(y,x) */
depends(y, x)$
print("  Με depends(y,x):", solve(diff(x^2 + y^2 - 25, x) = 0, 'diff(y,x)))$
remove(y, dependency)$

/* ── Ζ. Προαιρετικό: λογαριθμική παραγώγιση (x^x)' ──────── */
print("")$
print("════ Ζ. Λογαριθμική παραγώγιση: (x^x)' ════")$

assume(x > 0)$                          /* ώστε να ορίζεται το ln x */
print("  (x^x)' =", ratsimp(diff(x^x, x)), "   (αναμ. x^x (ln x + 1))")$
print("  Έλεγχος (πρέπει 0):", ratsimp(diff(x^x, x) - x^x*(log(x) + 1)))$
forget(x > 0)$

print("")$
print("Ολοκληρώθηκε το Κεφάλαιο 10.")$
