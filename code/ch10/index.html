   Matzakos, N. (2026). Elements of Linear Algebra, Differential & Integral Calculus. NewTech Publications.

   COMMANDS USED:
     diff(f, x)                     -> f'(x)
     diff(f, x, n)                  -> nth derivative
     at(diff(f,x), x=a)             -> value of the derivative at a
     subst(a, x, expr)              -> substitution x -> a
     trigsimp(...) , ratsimp(...)   -> simplification
     limit((f(x+h)-f(x))/h, h, 0)   -> derivative from the DEFINITION

   Run with: wxMaxima (File/Batch) or
                 maxima -b maxima_paragogos.mac
   ============================================================ */

/* ── A. Basic differentiation rules ─────────────────────── */
print("════ A. Basic rules ════")$

fA1 : 3*x^4 - 5*x^2 + 2*x - 7$
fA2 : sin(x)$
fA3 : exp(3*x)$
fA4 : log(x^2 + 1)$

print("(3x^4-5x^2+2x-7)' =", diff(fA1, x), "   (expected 12x^3-10x+2)")$
print("(sin x)'          =", diff(fA2, x), "   (expected cos x)")$
print("(e^(3x))'         =", diff(fA3, x), "   (expected 3e^(3x))")$
print("(ln(x^2+1))'      =", diff(fA4, x), "   (expected 2x/(x^2+1))")$

/* Check by hand: power rule, chain rule, logarithm. */
print("Check (difference from the expected value, must be 0):",
      ratsimp(diff(fA1,x) - (12*x^3-10*x+2)),
      ratsimp(diff(fA2,x) - cos(x)),
      ratsimp(diff(fA3,x) - 3*exp(3*x)),
      ratsimp(diff(fA4,x) - 2*x/(x^2+1)))$

/* ── B. Product, quotient, chain ────────────────────────── */
print("")$
print("════ B. Product — Quotient — Chain ════")$

/* Product rule: (uv)' = u'v + uv' */
gB : x^2*sin(x)$
print("(x^2 sin x)' =", diff(gB, x), "   (expected 2x sin x + x^2 cos x)")$

/* Quotient rule: (u/v)' = (u'v - uv')/v^2 */
hB : (x^3 + 1)/(x^2 - 1)$
print("((x^3+1)/(x^2-1))' =", factor(ratsimp(diff(hB, x))),
      "   (expected x(x-2)/(x-1)^2)")$

/* Chain rule: [f(g(x))]' = f'(g(x)) g'(x) */
kB : sin(x^2 + 1)$
print("[sin(x^2+1)]' =", diff(kB, x), "   (expected 2x cos(x^2+1))")$

print("Checks (must be 0,0,0):",
      ratsimp(diff(gB,x) - (2*x*sin(x) + x^2*cos(x))),
      ratsimp(diff(hB,x) - x*(x-2)/(x-1)^2),
      ratsimp(diff(kB,x) - 2*x*cos(x^2+1)))$

/* ── C. Higher order: f(x) = e^x cos x ──────────────────── */
print("")$
print("════ C. Higher-order derivatives: f(x)=e^x cos(x) ════")$

fC : exp(x)*cos(x)$
for n:1 thru 4 do
    print("  f^(", n, ")(x) =", expand(diff(fC, x, n)))$

/* The required identity: f^(4) = -4 f  */
print("  f^(4) + 4 f =", expand(diff(fC, x, 4) + 4*fC),
      "  =>  f^(4) = -4f  (verified)")$

/* Higher order for sin as well: a cycle of period 4 */
print("  For comparison, sin^(n)(x):")$
for n:1 thru 4 do print("    n =", n, "->", diff(sin(x), x, n))$

/* ── D. Tangent line: f(x)=x^3-2x+1 at x0=1 ─────────────── */
print("")$
print("════ D. Tangent to f(x)=x^3-2x+1 at x0=1 ════")$

fD : x^3 - 2*x + 1$
x0 : 1$
y0 : subst(x0, x, fD)$
m  : at(diff(fD, x), x = x0)$          /* at(): value of the derivative at x0 */
tang : expand(y0 + m*(x - x0))$

print("  f(1) =", y0, "   f'(x) =", diff(fD,x), "   f'(1) =", m)$
print("  Tangent: y =", tang, "   (expected y = x - 1)")$
print("  Check (must be 0):", ratsimp(tang - (x - 1)))$

/* Plot of f, f', f'' and of the tangent line on [-2, 2.5] */
plot2d([fD, diff(fD,x), diff(fD,x,2), tang], [x, -2, 2.5],
       [legend, "f", "f'", "f''", "tangent at x0=1"],
       [title, "f(x)=x^3-2x+1 : f, f', f'' and the tangent"])$

/* ── E. The derivative from the DEFINITION (a limit) ────── */
print("")$
print("════ E. Definition: f'(x) = lim_{h->0} (f(x+h)-f(x))/h ════")$

/* f(x) = x^2 */
print("  x^2 : difference quotient =", ratsimp(((x+h)^2 - x^2)/h))$
print("        limit h->0          =", limit(((x+h)^2 - x^2)/h, h, 0),
      "   (diff: ", diff(x^2, x), ")")$

/* f(x) = sqrt(x) */
print("  sqrt(x) : limit h->0     =", limit((sqrt(x+h) - sqrt(x))/h, h, 0),
      "   (diff: ", diff(sqrt(x), x), ")")$

/* One more: f(x)=sin x — the definition gives cos x */
print("  sin(x)  : limit h->0     =", limit((sin(x+h) - sin(x))/h, h, 0))$

/* One-sided limits: |x| is NOT differentiable at 0 */
print("")$
print("  |x| at 0 — from the right:", limit((abs(0+h) - abs(0))/h, h, 0, plus))$
print("  |x| at 0 — from the left :", limit((abs(0+h) - abs(0))/h, h, 0, minus))$
print("  The one-sided limits differ => there is no derivative at 0.")$

/* ── F. Implicit differentiation: x^2 + y^2 = 25 ────────── */
print("")$
print("════ F. Implicit differentiation: x^2 + y^2 = 25 ════")$

/* Implicit function theorem: dy/dx = -F_x / F_y  (partial derivatives).
   NOTE: without depends(y,x) diff is partial — which is what we want. */
eq : x^2 + y^2 - 25$
dy_dx : ratsimp(-diff(eq, x)/diff(eq, y))$
print("  F_x =", diff(eq,x), "   F_y =", diff(eq,y))$
print("  dy/dx =", dy_dx, "   (expected -x/y)")$
print("  Check (must be 0):", ratsimp(dy_dx + x/y))$

/* Alternatively, with depends(y,x) we solve for 'diff(y,x) */
depends(y, x)$
print("  With depends(y,x):", solve(diff(x^2 + y^2 - 25, x) = 0, 'diff(y,x)))$
remove(y, dependency)$

/* ── G. Optional: logarithmic differentiation (x^x)' ────── */
print("")$
print("════ G. Logarithmic differentiation: (x^x)' ════")$

assume(x > 0)$                          /* so that ln x is defined */
print("  (x^x)' =", ratsimp(diff(x^x, x)), "   (expected x^x (ln x + 1))")$
print("  Check (must be 0):", ratsimp(diff(x^x, x) - x^x*(log(x) + 1)))$
forget(x > 0)$

print("")$
print("Chapter 10 completed.")$
