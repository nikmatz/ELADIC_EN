/* ============================================================
   maxima_diag.mac
   Chapter 6 — Diagonalization, PDP⁻¹, Orthogonal Diagonalization,
                 Quadratic Forms
   Matzakos, N. (2026). Elements of Linear Algebra, Differential & Integral Calculus. NewTech Publications.
   ============================================================

   COMMANDS USED:
     load("eigen")             → eigenvalue package
     eigenvectors(A)           → eigenvalues, multiplicities, eigenvectors
     invert(P)                 → inverse of the eigenvector matrix
     transpose(Q)              → for an orthonormal Q we have Q⁻¹ = Qᵀ
     ratsimp(P . D . invert(P))→ simplification of the product
     nullspace(B - 3*ident(2)) → eigenspace of the double eigenvalue
     gramschmidt(M)            → Gram–Schmidt orthogonalization
     A^^n                      → nth power of a matrix

   Book activity: (a) diagonalization of A and A¹⁰,
     (b) why B is not diagonalizable, (c) the spectral theorem for S,
     (d) the quadratic form xᵀSx and its classification.

   Run with: wxMaxima (Load/File) or
                 maxima -b maxima_diag.mac
   ============================================================ */

load("eigen")$

/* Helper: equality of matrices through their difference */
mat_equal(M, N) := is( ratsimp(M - N) = zeromatrix(length(M), length(M[1])) )$

/* Helper: the product xᵀ·S·x may be returned either as a 1×1
   matrix or already as a scalar — here we always take the scalar. */
scal(M) := if matrixp(M) then M[1][1] else M$

/* Helper: list of the eigenvalues (with multiplicities) and list
   of the eigenvectors as ROWS, exactly as eigenvectors() returns them.
   We do NOT write eigenvectors "by hand": we get them from Maxima. */
eig_pairs(M) := block([ev, vals, mults, vecs, lv, lw, i, k],
    ev : eigenvectors(M),
    vals : ev[1][1], mults : ev[1][2], vecs : ev[2],
    lv : [], lw : [],
    for i : 1 thru length(vals) do
        for k : 1 thru length(vecs[i]) do (
            lv : endcons(vals[i], lv),
            lw : endcons(vecs[i][k], lw)
        ),
    [lv, lw])$

/* ── A. (a) Diagonalization of A: P, D, PDP⁻¹ = A, A¹⁰ ──── */
print("════ A. (a) Diagonalization A = P·D·P⁻¹ ════");

A : matrix([5, 4],
           [1, 2]);
print("A ="); print(A);

print("eigenvectors(A) ="); print(eigenvectors(A));

pairsA : eig_pairs(A)$
lamA : pairsA[1]$        /* eigenvalues */
vecA : pairsA[2]$        /* eigenvectors as rows */
print("Eigenvalues lam =", lamA, "  [Expected {6, 1}]");
print("Eigenvectors (rows) =", vecA);

P : transpose(apply(matrix, vecA));
D : apply(diag_matrix, lamA);
print("P = [v₁|v₂] ="); print(P);
print("D = diag(lam1,lam2) ="); print(D);

print("det(P) =", determinant(P),
      " ≠ 0 → the eigenvectors are linearly independent:",
      is(determinant(P) # 0));

print("P⁻¹ ="); print(invert(P));
print("P·D·P⁻¹ ="); print(ratsimp(P . D . invert(P)));
print("P·D·P⁻¹ = A ;  Check:", mat_equal(P . D . invert(P), A));

/* A¹⁰ = P·D¹⁰·P⁻¹ */
n_pow : 10;
D10 : apply(diag_matrix, map(lambda([t], t^n_pow), lamA));
A10 : ratsimp(P . D10 . invert(P));
print("D¹⁰ ="); print(D10);
print("A¹⁰ = P·D¹⁰·P⁻¹ ="); print(A10);
print("  [Expected: matrix([48372941,48372940],[12093235,12093236])]");
print("Comparison with the direct power A^^10 ;  Check:",
      mat_equal(A10, A^^n_pow));

/* ── B. (b) B is not diagonalizable ─────────────────────── */
print("════ B. (b) B = [[3,1],[0,3]] — not diagonalizable ════");

B : matrix([3, 1],
           [0, 3]);
print("B ="); print(B);
print("charpoly(B, lam) =", factor(expand(charpoly(B, lam))),
      "  [Expected (lam-3)²]");
print("eigenvalues(B) =", eigenvalues(B),
      "  → lam = 3 with algebraic multiplicity m_a = 2");

B3 : B - 3*ident(2);
print("B - 3·I ="); print(B3);
nsB : nullspace(B3);
print("nullspace(B - 3*ident(2)) =", nsB);
m_g : length(args(nsB));
m_a : eigenvalues(B)[2][1];
print("geometric multiplicity m_g = 2 - rank(B-3I) =", 2 - rank(B3),
      " = ", m_g);
print("algebraic multiplicity m_a =", m_a);
print("m_g < m_a ;  Check:", is(m_g < m_a));
print("Conclusion: is B diagonalizable?", is(m_g = 2));
print("Why: diagonalization requires 2 linearly independent");
print("eigenvectors, so that P = [v₁|v₂] is invertible.");
print("The eigenspace of lam=3 is ONE-DIMENSIONAL, so there is no");
print("basis of R² consisting of eigenvectors (defective eigenvalue).");

/* ── C. (c) Spectral theorem for the symmetric S ───────── */
print("════ C. (c) Orthogonal Diagonalization S = Q·Lambda·Qᵀ ════");

S : matrix([4, 2],
           [2, 1]);
print("S ="); print(S);
print("S = Sᵀ ;  Check:", mat_equal(S, transpose(S)));

print("eigenvectors(S) ="); print(eigenvectors(S));

pairsS : eig_pairs(S)$
lamS : pairsS[1]$
vecS : pairsS[2]$
print("Eigenvalues lam =", lamS, "  [Expected {0, 5}]");
print("Eigenvectors (rows, from Maxima) =", vecS);

/* Test of orthogonality BEFORE Gram–Schmidt:
   for a symmetric matrix with distinct eigenvalues they are already orthogonal */
print("v₁ · v₂ =", ratsimp(vecS[1] . vecS[2]),
      " → already orthogonal:", is(ratsimp(vecS[1] . vecS[2]) = 0));

/* Gram–Schmidt and normalization */
gs  : gramschmidt(apply(matrix, vecS))$
gsn : map(lambda([u], ratsimp(u / sqrt(u . u))), gs)$
print("Gram–Schmidt →", gs);
print("Orthonormal →", gsn);

Q  : transpose(apply(matrix, gsn));
LS : apply(diag_matrix, lamS);
print("Q ="); print(Q);
print("Q (decimal) ="); print(float(Q));
print("Lambda ="); print(LS);

print("QᵀQ ="); print(ratsimp(transpose(Q) . Q));
print("QᵀQ = I₂ ;  Check:", mat_equal(transpose(Q) . Q, ident(2)));
print("Q⁻¹ = Qᵀ ;  Check:", mat_equal(invert(Q), transpose(Q)));
print("det(Q) =", ratsimp(determinant(Q)), " (±1 for an orthogonal matrix)");

print("Q·Lambda·Qᵀ ="); print(ratsimp(Q . LS . transpose(Q)));
print("Q·Lambda·Qᵀ = S ;  Check:", mat_equal(Q . LS . transpose(Q), S));
print("  [Spectral theorem: every symmetric matrix is Q·Lambda·Qᵀ ✓]");

/* ── D. (d) Quadratic form Q(x) = xᵀSx and classification ─ */
print("════ D. (d) Quadratic Form Q(x) = xᵀ·S·x ════");

xv : matrix([x], [y]);
qf : expand( scal(transpose(xv) . S . xv) );
print("Q(x,y) = xᵀSx =", qf);
print("  in factored form:", factor(qf));
print("  [Expected: 4x² + 4xy + y² = (2x+y)² ✓]");

/* In the new (principal) coordinates x = Q·u the form becomes sum(lam_i·u_i²) */
uv : matrix([u1], [u2]);
qf_diag : expand( scal(transpose(uv) . LS . uv) );
print("In principal axes (x = Q·u): Q =", qf_diag, "  [= sum(lam_i·u_i²)]");
print("Identity Q(Q·u) = sum(lam_i·u_i²) ;  Check:",
      is( ratsimp( expand(scal(transpose(Q . uv) . S . (Q . uv))) - qf_diag ) = 0 ));

/* Classification EXCLUSIVELY from the signs of the eigenvalues */
n_pos : length(sublist(lamS, lambda([t], is(t > 0))))$
n_neg : length(sublist(lamS, lambda([t], is(t < 0))))$
n_zero : length(sublist(lamS, lambda([t], is(t = 0))))$
print("Eigenvalues:", lamS, " → positive:", n_pos,
      ", negative:", n_neg, ", zero:", n_zero);

classification :
  if n_neg = 0 and n_zero = 0 then "positive definite"
  elseif n_pos = 0 and n_zero = 0 then "negative definite"
  elseif n_neg = 0 then "positive semidefinite"
  elseif n_pos = 0 then "negative semidefinite"
  else "indefinite"$
print("Classification of Q:", classification);

/* Verification of the conclusion. Rayleigh quotient: for ‖x‖ = 1 we have
   lam_min ≤ Q(x) ≤ lam_max, with the extremes at the corresponding eigenvectors. */
samples : float( makelist( subst([x = cos(2*%pi*k/360), y = sin(2*%pi*k/360)],
                                 qf), k, 0, 359) )$
print("Sample of 360 points of the unit circle:");
print("  min Q =", lmin(samples), " | max Q =", lmax(samples));
print("  No negative value:", is(lmin(samples) >= 0),
      " | max ≤ lam_max = 5:", is(lmax(samples) <= 5));

/* Exact extremes: at the columns of Q (orthonormal eigenvectors) */
for j : 1 thru 2 do block([cj, qj],
    cj : col(Q, j),
    qj : ratsimp( scal(transpose(cj) . S . cj) ),
    print("  Q(q", j, ") =", qj, " = lam", j, " =", lamS[j],
          " | Equality:", is(qj = lamS[j]))
);
print("There is x ≠ 0 with Q(x) = 0 (e.g. x = (1,-2)ᵀ):",
      subst([x = 1, y = -2], qf),
      "→ hence SEMI-definite and not (strictly) positive definite.");

print("════ End ════");
