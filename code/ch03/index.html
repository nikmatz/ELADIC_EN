   COMMANDS USED:
     solve([eq1,eq2,...],[x,y,...])  → solve a system symbolically
     linsolve([eq1,...],[x,y,...])   → linear system (more efficient)
     echelon(A)                      → row echelon form (Maxima does NOT provide rref)
     rank(A)                         → rank of a matrix
     determinant(A)                  → determinant (Cramer)
     augmented_matrix(A,b)           → augmented matrix [A|b]

   Run with: wxMaxima (Load/File) or
                 maxima -b maxima_linsys.mac
   ============================================================ */

/* ── Part A: System with a Unique Solution ──────────────── */
print("════ A. Unique Solution ════");

/* 2x + y - z  =  8
   -3x - y + 2z = -11
   -2x +  y + 2z = -3   */

eq1 :  2*x +   y -   z =  8;
eq2 : -3*x -   y + 2*z = -11;
eq3 :  -2*x +   y + 2*z = -3;

sol : linsolve([eq1, eq2, eq3], [x, y, z]);
print("Solution:", sol);

/* Check */
x0 : rhs(sol[1]);
y0 : rhs(sol[2]);
z0 : rhs(sol[3]);
print("Check eq1:", ev(lhs(eq1), x=x0, y=y0, z=z0), "=", rhs(eq1));
print("Check eq2:", ev(lhs(eq2), x=x0, y=y0, z=z0), "=", rhs(eq2));

/* By means of the row echelon form */
print("");
A1 : matrix([2, 1, -1], [-3, -1, 2], [-2, 1, 2]);
b1 : matrix([8], [-11], [-3]);
Aug1 : addcol(A1, b1);
print("Augmented [A|b] ="); print(Aug1);
print("Echelon([A|b]) ="); print(echelon(Aug1));

/* ── Part B: System with No Solution (Inconsistent) ────── */
print("════ B. Inconsistent System ════");

/*  x + y  = 3
    2x + 2y = 5  (inconsistent: 2·(eq1) → 2x+2y=6 ≠ 5) */
A2 : matrix([1, 1], [2, 2]);
b2 : matrix([3], [5]);
Aug2 : addcol(A2, b2);
print("Echelon([A|b]) ="); print(echelon(Aug2));
print("rank(A) =", rank(A2), "  rank([A|b]) =", rank(Aug2));
print("→ rank(A) < rank([A|b])  ⇒  INCONSISTENT system (0 solutions)");

/* ── Part C: System with Infinitely Many Solutions ─────── */
print("════ C. Infinitely Many Solutions ════");

/*  x  + 2y - z = 3
    2x + 4y - 2z = 6   (= 2·eq1 → dependent) */
A3 : matrix([1, 2, -1], [2, 4, -2]);
b3 : matrix([3], [6]);
Aug3 : addcol(A3, b3);
print("Echelon([A|b]) ="); print(echelon(Aug3));
print("rank(A) =", rank(A3), "  n =", 3);
print("→ rank(A) = rank([A|b]) < n  ⇒  INFINITELY MANY solutions");
print("  Free variables:", 3 - rank(A3));

sol3 : linsolve([x + 2*y - z = 3, 2*x + 4*y - 2*z = 6], [x, y, z]);
print("Solution (SymPy-style):", sol3);

/* ── Part D: Cramer's Rule ───────────────────────────────── */
print("════ D. Cramer's Rule ════");
/* Only for square systems with det ≠ 0 */

A_cr : matrix([2, 1], [5, 3]);
b_cr : matrix([4], [7]);
det_A : determinant(A_cr);
print("det(A) =", det_A);

/* Column replacement for each unknown */
A_x : copymatrix(A_cr);
A_x[1][1] : b_cr[1][1];   A_x[2][1] : b_cr[2][1];
A_y : copymatrix(A_cr);
A_y[1][2] : b_cr[1][1];   A_y[2][2] : b_cr[2][1];

x_cr : determinant(A_x) / det_A;
y_cr : determinant(A_y) / det_A;
print("x = det(Ax)/det(A) =", x_cr);
print("y = det(Ay)/det(A) =", y_cr);

/* Check */
print("Check:", 2*x_cr + y_cr, "=", 4, "  and  ", 5*x_cr + 3*y_cr, "=", 7);

/* ── Part E: Homogeneous System ──────────────────────────── */
print("════ E. Homogeneous System Ax = 0 ════");

A5 : matrix([1, -2,  1],
            [2, -3,  1],
            [0,  1, -1]);
b5 : matrix([0], [0], [0]);
Aug5 : addcol(A5, b5);
print("Echelon([A|0]) ="); print(echelon(Aug5));
print("rank(A) =", rank(A5), "  nullity =", 3 - rank(A5));
print("Nontrivial solutions (nullity =", 3-rank(A5), "> 0):");
print(nullspace(A5));

print("════ End ════");
