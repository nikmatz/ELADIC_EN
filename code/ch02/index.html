/* ============================================================
   maxima_matrices.mac
   Chapter 2 — Matrix Operations: Product, Inverse, LU
   Matzakos, N. (2026). Elements of Linear Algebra, Differential & Integral Calculus. NewTech Publications.
   ============================================================

   COMMANDS USED:
     A . B                 → matrix product (the "." operator)
     A + B,  k * A         → addition / multiplication by a scalar
     transpose(A)          → transpose matrix Aᵀ
     determinant(A)        → determinant det(A)
     invert(A)             → inverse matrix A⁻¹
     ident(n)              → n×n identity matrix
     zeromatrix(m,n)       → m×n zero matrix
     echelon(A)            → row echelon form (Maxima does NOT have rref)
     triangularize(A)      → upper triangularization (without division)
     lu_factor / get_lu_factors  → decomposition A = P·L·U (linearalgebra)

   Book activity: (a) operations, (b) determinants,
                          (c) the inverse, (d) solving A·x = b.

   Run with: wxMaxima (Load/File) or
                 maxima -b maxima_matrices.mac
   ============================================================ */

load(linearalgebra)$

/* Helper: test equality of matrices through their difference */
mat_equal(M, N) := is( ratsimp(M - N) = zeromatrix(length(M), length(M[1])) )$

/* Data of the activity */
A : matrix([1, 2, 3],
           [0, 1, 4],
           [5, 6, 0]);

B : matrix([2, -1, 0],
           [1,  3, 1],
           [0,  2, 4]);

b_vec : matrix([1], [2], [3]);

print("A ="); print(A);
print("B ="); print(B);
print("b ="); print(b_vec);

/* ── A. (a) A+B, 3A, A·B, B·A  and  A·B ≠ B·A ──────────── */
print("════ A. (a) Matrix operations ════");

print("A + B ="); print(A + B);
print("  [Expected: matrix([3,1,3],[1,4,5],[5,8,4]) ✓]");

print("3·A ="); print(3 * A);
print("  [Expected: matrix([3,6,9],[0,3,12],[15,18,0]) ✓]");

AB : A . B;
BA : B . A;
print("A·B ="); print(AB);
print("  [Expected: matrix([4,11,14],[1,11,17],[16,13,6]) ✓]");
print("B·A ="); print(BA);
print("  [Expected: matrix([2,3,2],[6,11,15],[20,26,8]) ✓]");

print("A·B - B·A ="); print(AB - BA);
print("A·B = B·A ;  Check:", mat_equal(AB, BA));
print("  → false: matrix multiplication is NOT commutative");

/* Transpose (also useful below) */
print("Aᵀ ="); print(transpose(A));
print("(Aᵀ)ᵀ = A ;  Check:", mat_equal(transpose(transpose(A)), A));
print("(A·B)ᵀ = Bᵀ·Aᵀ ;  Check:",
      mat_equal(transpose(AB), transpose(B) . transpose(A)));

/* ── B. (b) Determinants and det(A·B) = det(A)·det(B) ──── */
print("════ B. (b) Determinants ════");

det_A : determinant(A);
det_B : determinant(B);
det_AB : determinant(AB);

print("det(A) =", det_A, "  [Expected 1]");
print("det(B) =", det_B, "  [Expected 24]");
print("det(A·B) =", det_AB, "  [Expected 24]");
print("det(A)·det(B) =", det_A * det_B);
print("det(A·B) = det(A)·det(B) ;  Check:", is(det_AB = det_A * det_B));

/* ── C. (c) The inverse A⁻¹ and A·A⁻¹ = I₃ ─────────────── */
print("════ C. (c) The inverse ════");

if det_A # 0 then (
    A_inv : invert(A),
    print("A⁻¹ ="),
    print(A_inv),
    print("  [Expected: matrix([-24,18,5],[20,-15,-4],[-5,4,1]) ✓]"),
    print("A·A⁻¹ ="),
    print(ratsimp(A . A_inv)),
    print("A·A⁻¹ = I₃ ;  Check:", mat_equal(A . A_inv, ident(3))),
    print("A⁻¹·A = I₃ ;  Check:", mat_equal(A_inv . A, ident(3))),
    print("det(A⁻¹) =", determinant(A_inv), " = 1/det(A) ;  Check:",
          is(determinant(A_inv) = 1/det_A))
) else (
    A_inv : 'undefined,
    print("det(A) = 0  →  A is not invertible")
);

/* ── D. (d) Solving A·x = b with invert(A) ─────────────── */
print("════ D. (d) Solving A·x = b ════");

x_sol : ratsimp(A_inv . b_vec);
print("x = A⁻¹·b ="); print(x_sol);
print("  [Expected: x = (27, -22, 6)ᵀ ✓]");

print("Check A·x ="); print(ratsimp(A . x_sol));
print("A·x = b ;  Check:", mat_equal(A . x_sol, b_vec));

/* Comparison with the direct solver linsolve */
sol_ls : linsolve([   x1 + 2*x2 + 3*x3 = 1,
                            x2 + 4*x3 = 2,
                   5*x1 + 6*x2        = 3], [x1, x2, x3]);
print("linsolve →", sol_ls, "  [same result ✓]");

/* ── E. Supplement: row echelon form and LU decomposition ─ */
print("════ E. Row echelon form & LU ════");

/* NOTE: Maxima does not provide rref(). The counterparts are
   echelon(A) (echelon form with leading 1s) or triangularize(A). */
print("echelon(A) ="); print(echelon(A));
print("triangularize(A) ="); print(triangularize(A));
print("rank(A) =", rank(A), "  [full rank 3, hence A is invertible ✓]");

/* LU decomposition: lu_factor + get_lu_factors (linearalgebra package).
   With generalring we get EXACT (rational) coefficients. */
lu_data : lu_factor(A, generalring)$
lu_PLU  : get_lu_factors(lu_data)$
P_lu : lu_PLU[1];
L_lu : lu_PLU[2];
U_lu : lu_PLU[3];

print("P ="); print(P_lu);
print("L (lower triangular) ="); print(L_lu);
print("U (upper triangular) ="); print(U_lu);
print("P·L·U ="); print(ratsimp(P_lu . L_lu . U_lu));
print("P·L·U = A ;  Check:", mat_equal(P_lu . L_lu . U_lu, A));
print("det(A) = det(L)·det(U) =", determinant(L_lu) * determinant(U_lu),
      ";  Check:", is(determinant(L_lu)*determinant(U_lu) = det_A));

print("════ End ════");
